--- TITLE ---
Option/Result method on a constructor-inferred receiver lowers to the arilrt method (unwrapOr→UnwrapOr)

--- INPUT ---
import fmt

func main() {
  // Receiver type inferred directly from a `Some(...)` constructor (no
  // annotation, no signature) — its query/defaulting methods must still
  // lower to the exported arilrt spelling (builtins.md §Option methods).
  let o = Some(5)
  fmt.println(o.isSome(), o.isNone(), o.unwrapOr(-1))

  // Bare constructor literal as the receiver.
  fmt.println(Some(7).unwrapOr(-1))

  // `.map` on a constructor-inferred Option flows through the same path.
  let doubled = Some(10).map((x) => x * 2)
  fmt.println(doubled.unwrapOr(0))

  // Result via an explicitly-typed binding: the method resolves the same
  // way once the receiver's Result type is known.
  let r: Result<int, string> = Ok(3)
  fmt.println(r.isOk(), r.isErr(), r.unwrapOr(0))
}

--- GO ---
package main

import "fmt"

type Option[T any] struct {
	Tag uint8
	V   T
}

func OptionSome[T any](value T) Option[T] {
	return Option[T]{Tag: 1, V: value}
}
func OptionNone[T any]() Option[T] {
	return Option[T]{Tag: 0}
}
func (o Option[T]) String() string {
	if o.Tag == 1 {
		return fmt.Sprintf("Some(%v)", o.V)
	}
	return "None"
}
func (o Option[T]) IsSome() bool { return o.Tag == 1 }
func (o Option[T]) IsNone() bool { return o.Tag == 0 }
func (o Option[T]) UnwrapOr(fallback T) T {
	if o.Tag == 1 {
		return o.V
	}
	return fallback
}

type Result[T any, E any] struct {
	Tag uint8
	V   T
	E   E
}

func ResultOk[T any, E any](value T) Result[T, E] {
	return Result[T, E]{Tag: 0, V: value}
}
func ResultErr[T any, E any](err E) Result[T, E] {
	return Result[T, E]{Tag: 1, E: err}
}
func (r Result[T, E]) String() string {
	if r.Tag == 0 {
		return fmt.Sprintf("Ok(%v)", r.V)
	}
	return fmt.Sprintf("Err(%v)", r.E)
}
func (r Result[T, E]) IsOk() bool  { return r.Tag == 0 }
func (r Result[T, E]) IsErr() bool { return r.Tag == 1 }
func (r Result[T, E]) UnwrapOr(fallback T) T {
	if r.Tag == 0 {
		return r.V
	}
	return fallback
}
func OptionMap[T any, U any](o Option[T], f func(T) U) Option[U] {
	if o.Tag == 1 {
		return Option[U]{Tag: 1, V: f(o.V)}
	}
	return Option[U]{Tag: 0}
}
func main() {
	var o = OptionSome(5)
	fmt.Println(o.IsSome(), o.IsNone(), o.UnwrapOr(-1))
	fmt.Println(OptionSome(7).UnwrapOr(-1))
	var doubled = OptionMap(OptionSome(10), func(x int) int {
		return x * 2
	})
	fmt.Println(doubled.UnwrapOr(0))
	var r Result[int, string] = ResultOk[int, string](3)
	fmt.Println(r.IsOk(), r.IsErr(), r.UnwrapOr(0))
}
