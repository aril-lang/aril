--- TITLE ---
Closure result type inferred from the call parameter — a `Result<unit, string>`
body with `try`; the closure is its own try frame (CLOSURE-INFERENCE)

--- INPUT ---
import fmt

func want(cond: bool, msg: string): Result<unit, string> {
  if !cond { return Err(msg) }
  return Ok(())
}

func run(check: (int) => Result<unit, string>, n: int): bool {
  match check(n) {
    Ok(_) => { return true },
    Err(_) => { return false },
  }
}

func main() {
  let ok = run((n) => {
    try want(n > 0, "positive")
    return Ok(())
  }, 5)
  fmt.println(ok)
}
